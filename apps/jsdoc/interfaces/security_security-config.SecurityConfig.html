    Interface SecurityConfig

    Security headers configuration for Jen.js.

    Comprehensive configuration for all HTTP security headers, with sensible defaults and full customization options.

    Example

    // In jen.config.ts
    security: {
      headers: {
        csp: {
          enabled: true,
          directives: {
            defaultSrc: ["'self'"],
            scriptSrc: ["'self'", "'unsafe-inline'"]
          }
        },
        hsts: { enabled: true, maxAge: 31536000 },
        cors: { origins: ["https://example.com"] }
      }
    }
    interface SecurityConfig {
        csrf?: { cookieName?: string; enabled?: boolean; headerName?: string };
        headers?: {
            contentTypeOptions?: boolean;
            cors?: CORSConfig;
            csp?: CSPConfig;
            frameOptions?: FrameOptionsConfig;
            hsts?: HSTSConfig;
            permissionsPolicy?: PermissionsPolicyConfig;
            referrerPolicy?: ReferrerPolicy;
        };
        rateLimit?: {
            enabled?: boolean;
            maxRequests?: number;
            skipPaths?: string[];
            windowSeconds?: number;
        };
        validation?: { enabled?: boolean; maxBodySize?: number };
    }
